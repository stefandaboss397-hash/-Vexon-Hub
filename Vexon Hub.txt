local function new(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end

local function corner(parent, radius)
    return new("UICorner", {CornerRadius = UDim.new(0, radius)}, parent)
end

local function addStroke(parent, color, thickness)
    return new("UIStroke", {
        Color = color or C.stroke,
        Thickness = thickness or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    }, parent)
end

local function isPress(i)
    return i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch
end

local function isMove(i)
    return i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch
end

local function tween(obj, duration, props)
    pcall(function()
        TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props):Play()
    end)
end

local function atan2(y, x)
    if x > 0 then
        return math.atan(y / x)
    elseif x < 0 then
        return math.atan(y / x) + ((y >= 0) and math.pi or -math.pi)
    end
    return (y >= 0) and (math.pi / 2) or (-math.pi / 2)
end

local function drawLine(parent, x1, y1, x2, y2, th, color)
    local dx, dy = x2 - x1, y2 - y1
    local len = math.sqrt(dx * dx + dy * dy)
    return new("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.fromOffset((x1 + x2) / 2, (y1 + y2) / 2),
        Size = UDim2.fromOffset(len + 0.6, th),
        BackgroundColor3 = color,
        BorderSizePixel = 0,
        Rotation = math.deg(atan2(dy, dx)),
    }, parent)
end

local function drawPoly(parent, pts, th, color)
    for i = 1, #pts - 1 do
        drawLine(parent, pts[i][1], pts[i][2], pts[i + 1][1], pts[i + 1][2], th, color)
    end
end

local function drawBolt(parent, color)
    drawPoly(parent, {{10, 1}, {3, 10.5}, {8.5, 10.5}, {7, 17}, {15, 7}, {9.5, 7}, {10, 1}}, 1.5, color)
end

local function drawX(parent, color, size)
    local h = size / 2
    drawLine(parent, -h, -h, h, h, 2, color).Position = UDim2.new(0.5, 0, 0.5, 0)
    drawLine(parent, -h, h, h, -h, 2, color).Position = UDim2.new(0.5, 0, 0.5, 0)
end

local function drawCheck(parent, color)
    drawPoly(parent, {{4.5, 9.5}, {7.8, 12.8}, {13.5, 6}}, 2, color)
end

local function makeTabIcon(parent, kind)
    local box = new("Frame", {
        Position = UDim2.fromOffset(10, 9),
        Size = UDim2.fromOffset(14, 14),
        BackgroundTransparency = 1,
    }, parent)
    local strokes, fills = {}, {}

    local function ring(x, y, w, h)
        local f = new("Frame", {
            Position = UDim2.fromOffset(x, y),
            Size = UDim2.fromOffset(w, h),
            BackgroundTransparency = 1,
        }, box)
        corner(f, 99)
        strokes[#strokes + 1] = addStroke(f, C.dim, 1.5)
    end
    local function dot(x, y, s)
        local f = new("Frame", {
            Position = UDim2.fromOffset(x, y),
            Size = UDim2.fromOffset(s, s),
            BackgroundColor3 = C.dim,
            BorderSizePixel = 0,
        }, box)
        corner(f, 99)
        fills[#fills + 1] = f
    end

    if kind == "aim" then
        ring(0, 0, 14, 14); dot(5.5, 5.5, 3)
    elseif kind == "silent" then
        ring(0, 0, 14, 14); ring(4, 4, 6, 6)
    elseif kind == "eye" then
        ring(0, 3, 14, 8); dot(5, 5, 4)
    elseif kind == "player" then
        ring(4, 0, 6, 6)
        local body = new("Frame", {
            Position = UDim2.fromOffset(1, 8),
            Size = UDim2.fromOffset(12, 6),
            BackgroundTransparency = 1,
        }, box)
        corner(body, 4)
        strokes[#strokes + 1] = addStroke(body, C.dim, 1.5)
    elseif kind == "trigger" then
        ring(0, 0, 14, 14); ring(3.5, 3.5, 7, 7); dot(6, 6, 2)
    else
        ring(0, 0, 14, 14); dot(3, 4, 2.5); dot(8, 3.5, 2.5); dot(6, 8, 2.5)
    end

    return function(color)
        for _, s in ipairs(strokes) do s.Color = color end
        for _, f in ipairs(fills) do f.BackgroundColor3 = color end
    end
end

                    valLabel.Text = string.format(inc < 0.1 and "%.2f" or "%.1f", value)

                local rel = math.clamp((x - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
                local v = min + (max - min) * rel
                v = math.floor(v / inc + 0.5) * inc
                v = math.clamp(v, min, max)
                if v ~= value then
                    value = v
                    render()
                    if cb then pcall(cb, v) end
                end
            end

            local dragging = false
            hit.InputBegan:Connect(function(i)
                if isPress(i) then dragging = true; setFromX(i.Position.X) end

            UserInputService.InputChanged:Connect(function(i)

local function isTeammate(player)
    local a, b = localPlayer.Team, player.Team
    return a ~= nil and b ~= nil and a == b
end

local visParams = RaycastParams.new()
visParams.FilterType = Enum.RaycastFilterType.Exclude
visParams.IgnoreWater = true

local function partVisible(part)
    local char = localPlayer.Character
    if not part or not part.Parent then return false end
    local filter = {part.Parent}
    if char then filter[2] = char end
    visParams.FilterDescendantsInstances = filter
    local origin = camera.CFrame.Position
    return workspace:Raycast(origin, part.Position - origin, visParams) == nil
end

local function findTarget(visibleCheck, teamCheck, partName, fov)
    local vp = camera.ViewportSize
    local center = Vector2.new(vp.X / 2, vp.Y / 2)
    local best, bestDist = nil, fov

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer and not (teamCheck and isTeammate(player)) then
            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local part = char:FindFirstChild(partName) or char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
                if part then
                    local sp, onScreen = camera:WorldToViewportPoint(part.Position)
                    if onScreen then
                        local dist = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                        if dist <= bestDist then
                            if not visibleCheck or partVisible(part) then
                                best, bestDist = part, dist
                            end
                        end
                    end
                end
            end
        end
    end
    return best
end

    end)
end

local silentTarget = nil
local silentHook = nil
local lastFire = 0

local function installSilentHook()
    if silentHook ~= nil then return silentHook end
    silentHook = false
    if not (hookmetamethod and getnamecallmethod) then return false end
    local ok = pcall(function()
        local old
        local function hook(self, ...)
            local method = getnamecallmethod()
            if method == "Raycast" and S.silent and silentTarget and self == workspace and not (checkcaller and checkcaller()) then
                local args = {...}
                local origin, direction = args[1], args[2]
                if typeof(origin) == "Vector3" and typeof(direction) == "Vector3" then
                    if direction.Magnitude > 150 and (origin - camera.CFrame.Position).Magnitude < 12 then
                        args[2] = (silentTarget.Position - origin).Unit * direction.Magnitude
                        return old(self, table.unpack(args, 1, select("#", ...)))
                    end
                end
            end
            return old(self, ...)
        end
        old = hookmetamethod(game, "__namecall", newcclosure and newcclosure(hook) or hook)
    end)
    silentHook = ok
    return ok
end

local function silentStep()
    if not S.silent then
        silentTarget = nil
        return
    end
    silentTarget = findTarget(S.sVisible, S.sTeam, S.part, S.fovSize)
    if S.autoFire and silentTarget and tick() - lastFire > 0.12 then
        lastFire = tick()
        clickScreen()
    end
end

local lastTrigger = 0

local function triggerStep()
    if not S.trigger then return end
    if tick() - lastTrigger < math.max(S.fireDelay, 0.05) then return end

    local vp = camera.ViewportSize
    local center = Vector2.new(vp.X / 2, vp.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer and not (S.tTeam and isTeammate(player)) then
            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local head = char and char:FindFirstChild("Head")
            if hum and hum.Health > 0 and hrp and head then
                local topV, onTop = camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.6, 0))
                local botV, onBot = camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
                if onTop and onBot then
                    local h = math.abs(topV.Y - botV.Y)
                    local w = h / 2.2
                    local cx = (topV.X + botV.X) / 2
                    local yMin = math.min(topV.Y, botV.Y)
                    local yMax = math.max(topV.Y, botV.Y)
                    if math.abs(center.X - cx) <= w / 2 and center.Y >= yMin and center.Y <= yMax then
                        if not S.tVisible or partVisible(hrp) then
                            lastTrigger = tick()
                            clickScreen()
                            return
                        end
                    end
                end
            end
        end
    end
end

local espObjs = {}
local espHl = {}
local espActive = false

local function safeRemove(obj)
    if obj and obj.Remove then pcall(function() obj:Remove() end) end
end

local function removePlayerEsp(player)
    local o = espObjs[player]
    if o then
        for _, l in ipairs(o.box) do safeRemove(l) end
        for _, l in ipairs(o.skel) do safeRemove(l) end
        safeRemove(o.tracer)
        espObjs[player] = nil
    end
    if espHl[player] then
        pcall(function() espHl[player]:Destroy() end)
        espHl[player] = nil
    end
end

local function clearEsp()
    local list = {}
    for player, _ in pairs(espObjs) do list[#list + 1] = player end
    for player, _ in pairs(espHl) do list[#list + 1] = player end
    for _, player in ipairs(list) do removePlayerEsp(player) end
end

local function newEspObjs()
    local o = {box = {}, skel = {}, tracer = nil}
    for i = 1, 12 do
        local l = Drawing.new("Line")
        l.Visible = false; l.Thickness = 1
        o.box[i] = l
    end
    for i = 1, 14 do
        local l = Drawing.new("Line")
        l.Visible = false; l.Thickness = 1
        o.skel[i] = l
    end
    local t = Drawing.new("Line")
    t.Visible = false; t.Thickness = 1
    o.tracer = t
    return o
end

local function hideObjs(o)
    for _, l in ipairs(o.box) do l.Visible = false end
    for _, l in ipairs(o.skel) do l.Visible = false end
    o.tracer.Visible = false
end

local function espStep()
    local wantDraw = S.espBox or S.espSkeleton or S.espLines
    local wantHl = S.espOutline or S.espChams
    if not wantDraw and not wantHl then
        if espActive then
            clearEsp()
            espActive = false
        end
        return
    end
    espActive = true

    local camCF = camera.CFrame
    local vp = camera.ViewportSize

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer then
            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local hrp = char and char:FindFirstChild("HumanoidRootPart")

            if not hum or hum.Health <= 0 or not hrp then
                removePlayerEsp(player)
            else
                local visible = partVisible(hrp)
                local mate = isTeammate(player)
                local colName
                if mate then
                    colName = visible and S.colVisTeam or S.colInvTeam
                else
                    colName = visible and S.colVisEnemy or S.colInvEnemy
                end
                local color = colorOf(colName)

                local toTarget = hrp.Position - camCF.Position
                local cosA = math.clamp(camCF.LookVector:Dot(toTarget.Unit), -1, 1)
                local inFov = math.deg(math.acos(cosA)) <= S.espMaxFov / 2

                if wantHl and inFov then
                    local hl = espHl[player]
                    if not hl or hl.Parent ~= char then
                        if hl then pcall(function() hl:Destroy() end) end
                        hl = Instance.new("Highlight")
                        hl.Adornee = char
                        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        hl.Parent = char
                        espHl[player] = hl
                    end
                    hl.FillColor = color
                    hl.OutlineColor = color
                    hl.FillTransparency = S.espChams and 0.5 or 1
                    hl.OutlineTransparency = S.espOutline and 0 or 1
                elseif espHl[player] then
                    pcall(function() espHl[player]:Destroy() end)
                    espHl[player] = nil
                end

                if wantDraw and drawingOk and inFov then
                    if not espObjs[player] then
                        pcall(function() espObjs[player] = newEspObjs() end)
                    end
                    local o = espObjs[player]
                    if o then
                        local hv, onScreen = camera:WorldToViewportPoint(hrp.Position)
                        if onScreen then
                            if S.espBox then
                                local cf = hrp.CFrame
                                local hx, hy, hz = 1.5, 3, 1
                                local corners = {
                                    cf * CFrame.new(-hx, hy, -hz), cf * CFrame.new(hx, hy, -hz),
                                    cf * CFrame.new(hx, hy, hz), cf * CFrame.new(-hx, hy, hz),
                                    cf * CFrame.new(-hx, -hy, -hz), cf * CFrame.new(hx, -hy, -hz),
                                    cf * CFrame.new(hx, -hy, hz), cf * CFrame.new(-hx, -hy, hz),
                                }
                                local p2d = {}
                                local allOn = true
                                for i, c3 in ipairs(corners) do
                                    local sp, vis = camera:WorldToViewportPoint(c3.Position)
                                    if not vis then allOn = false end
                                    p2d[i] = Vector2.new(sp.X, sp.Y)
                                end
                                for idx, e in ipairs(BOX_EDGES) do
                                    local l = o.box[idx]
                                    if allOn then
                                        l.From = p2d[e[1]]
                                        l.To = p2d[e[2]]
                                        l.Color = color
                                        l.Visible = true
                                    else
                                        l.Visible = false
                                    end
                                end
                            else
                                for _, l in ipairs(o.box) do l.Visible = false end
                            end

                            if S.espSkeleton then
                                local bones = char:FindFirstChild("UpperTorso") and R15_BONES or R6_BONES
                                for idx = 1, 14 do
                                    local l = o.skel[idx]
                                    local pair = bones[idx]
                                    local shown = false
                                    if pair then
                                        local a = char:FindFirstChild(pair[1])
                                        local b = char:FindFirstChild(pair[2])
                                        if a and b then
                                            local pa, oa = camera:WorldToViewportPoint(a.Position)
                                            local pb, ob = camera:WorldToViewportPoint(b.Position)
                                            if oa and ob then
                                                l.From = Vector2.new(pa.X, pa.Y)
                                                l.To = Vector2.new(pb.X, pb.Y)
                                                l.Color = color
                                                l.Visible = true
                                                shown = true
                                            end
                                        end
                                    end
                                    if not shown then l.Visible = false end
                                end
                            else
                                for _, l in ipairs(o.skel) do l.Visible = false end
                            end

                            if S.espLines then
                                o.tracer.From = Vector2.new(vp.X / 2, vp.Y)
                                o.tracer.To = Vector2.new(hv.X, hv.Y)
                                o.tracer.Color = color
                                o.tracer.Visible = true
                            else
                                o.tracer.Visible = false
                            end
                        else
                            hideObjs(o)
                        end
                    end
                else
                    if espObjs[player] then hideObjs(espObjs[player]) end
                end
            end
        end
    end

local flyBV = nil

local function stopFly()
    if flyBV then
        pcall(function() flyBV:Destroy() end)
        flyBV = nil
    end
    local hum = getHum()
    if hum then pcall(function() hum.PlatformStand = false end) end
end

local function flyStep()
    if not S.fly then
        if flyBV then stopFly() end
        return
    end
    local root = getRoot()
    local hum = getHum()
    if not root or not hum then return end

    if not flyBV or flyBV.Parent ~= root then
        if flyBV then pcall(function() flyBV:Destroy() end) end
        flyBV = Instance.new("BodyVelocity")
        flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        flyBV.Velocity = Vector3.new(0, 0, 0)
        flyBV.Parent = root
    end
    hum.PlatformStand = true

    local cf = camera.CFrame
    local look = cf.LookVector
    local right = cf.RightVector
    local md = hum.MoveDirection
    local vel = Vector3.new(0, 0, 0)

    if md.Magnitude > 0 then
        local flatLook = Vector3.new(look.X, 0, look.Z)
        local flatRight = Vector3.new(right.X, 0, right.Z)
        if flatLook.Magnitude > 0 and flatRight.Magnitude > 0 then
            local f = md:Dot(flatLook.Unit)
            local r = md:Dot(flatRight.Unit)
            vel = (look * f + right * r) * S.flySpeed
        end
    end
    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
        vel = vel + Vector3.new(0, S.flySpeed, 0)
    end
    if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
        vel = vel - Vector3.new(0, S.flySpeed, 0)
    end
    flyBV.Velocity = vel
end

local function noclipStep()
    if not S.noclip then return end
    local char = getChar()
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.CanCollide then
            part.CanCollide = false
        end
    end
end

local function bhopStep()
    if not S.bhop then return end
    local hum = getHum()
    if not hum then return end
    if hum.MoveDirection.Magnitude > 0 and hum.FloorMaterial ~= Enum.Material.Air then
        hum:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end

local function speedStep()
    local hum = getHum()
    if not hum then return end
    if S.walk then
        hum.WalkSpeed = S.walkValue
    end
    if S.jump then
        hum.UseJumpPower = true
        hum.JumpPower = S.jumpValue
    end
end

local function resetWalk()
    local hum = getHum()
    if hum then pcall(function() hum.WalkSpeed = 16 end) end
end

local function resetJump()
    local hum = getHum()
    if hum then pcall(function() hum.UseJumpPower = true; hum.JumpPower = 50 end) end
end

local SKINS = {
    ["Phoenix Rifle"] = {mat = "Neon", colors = {Color3.fromRGB(235, 40, 30), Color3.fromRGB(255, 190, 40)}},
    ["Frost"] = {mat = "Ice", colors = {Color3.fromRGB(120, 200, 255), Color3.fromRGB(225, 245, 255)}},
    ["Toxic"] = {mat = "Neon", colors = {Color3.fromRGB(80, 255, 60), Color3.fromRGB(30, 120, 30)}},
    ["Void"] = {mat = "ForceField", colors = {Color3.fromRGB(150, 70, 255), Color3.fromRGB(40, 10, 90)}},
    ["Gold"] = {mat = "Metal", colors = {Color3.fromRGB(255, 200, 50), Color3.fromRGB(200, 140, 20)}},
    ["Rainbow"] = {mat = "Neon", rainbow = true},
}
local SKIN_NAMES = {"Phoenix Rifle", "Frost", "Toxic", "Void", "Gold", "Rainbow"}
local GUN_NAMES = {"All Guns", "Assault Rifle", "Handgun", "Shotgun", "Sniper", "Energy Rifle", "Exogun", "Burst Rifle", "Revolver", "Katana", "Chainsaw", "Fists"}

local skinActive = nil
local skinOrig = {}
local skinHue = 0

local function compact(s)
    return string.lower((string.gsub(s, "%s", "")))
end

local function gunMatches(model, gun)
    if gun == "All Guns" then return true end
    return string.find(compact(model.Name), compact(gun), 1, true) ~= nil
end

local function resetSkins()
    skinActive = nil
    local parts = {}
    for part, _ in pairs(skinOrig) do parts[#parts + 1] = part end
    for _, part in ipairs(parts) do
        local o = skinOrig[part]
        pcall(function()
            part.Material = o.Material
            part.Color = o.Color
        end)
        skinOrig[part] = nil
    end
end

local function skinStep(dt)
    if not skinActive then return end
    local def = SKINS[skinActive.skin]
    if not def then return end
    skinHue = skinHue + dt * 0.3

    local okMat, mat = pcall(function() return Enum.Material[def.mat] end)
    if not okMat or not mat then return end

    for _, model in ipairs(camera:GetChildren()) do
        if model:IsA("Model") and not model:FindFirstChildOfClass("Humanoid") and gunMatches(model, skinActive.gun) then
            local idx = 0
            for _, part in ipairs(model:GetDescendants()) do
                if part:IsA("BasePart") and part.Transparency < 1 then
                    if not skinOrig[part] then
                        skinOrig[part] = {Material = part.Material, Color = part.Color}
                    end
                    idx = idx + 1
                    part.Material = mat
                    if def.rainbow then
                        part.Color = Color3.fromHSV((skinHue + idx * 0.04) % 1, 1, 1)
                    else
                        part.Color = def.colors[(idx % #def.colors) + 1]
                    end
                end
            end
        end
    end
end

RunService.RenderStepped:Connect(function(dt)
    dt = dt or 0.016
    pcall(function() camera = workspace.CurrentCamera or camera end)
    pcall(fovStep)
    pcall(aimbotStep)
    pcall(silentStep)
    pcall(skinStep, dt)
end)

RunService.Heartbeat:Connect(function()
    pcall(espStep)
    pcall(flyStep)
    pcall(speedStep)
    pcall(bhopStep)
    pcall(triggerStep)
end)

RunService.Stepped:Connect(function()
    pcall(noclipStep)
end)
